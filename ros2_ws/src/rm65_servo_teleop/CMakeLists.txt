option(RM65_WITH_XR_SDK "Build the native XRoboToolkit input node" OFF)
if(RM65_WITH_XR_SDK)
  set(XR_SDK_ROOT "/opt/apps/roboticsservice/SDK" CACHE PATH "XR native SDK directory")
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    set(XR_ARCH arm64)
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    set(XR_ARCH x64)
  else()
    message(FATAL_ERROR "Unsupported native XR SDK architecture: ${CMAKE_SYSTEM_PROCESSOR}")
  endif()
  find_path(XR_INCLUDE PXREARobotSDK.h PATHS "${XR_SDK_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
  find_library(XR_LIBRARY PXREARobotSDK PATHS "${XR_SDK_ROOT}/${XR_ARCH}" REQUIRED NO_DEFAULT_PATH)
  add_executable(xr_sdk_node src/xr_sdk_node.cpp)
  target_include_directories(xr_sdk_node PRIVATE "${XR_INCLUDE}")
  ament_target_dependencies(xr_sdk_node rclcpp std_msgs)
  target_link_libraries(xr_sdk_node "${XR_LIBRARY}" Threads::Threads)
  set_target_properties(xr_sdk_node PROPERTIES INSTALL_RPATH "${XR_SDK_ROOT}/${XR_ARCH}")
  install(TARGETS xr_sdk_node DESTINATION lib/${PROJECT_NAME})
endif()
